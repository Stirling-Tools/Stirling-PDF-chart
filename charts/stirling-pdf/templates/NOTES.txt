{{- $url := trimSuffix "/" (.Values.rootPath | default "/") }}
** Please be patient while the chart is being deployed **

Get the stirlingpdf URL by running:
{{- if .Values.ingress.enabled }}
{{- range .Values.ingress.hosts }}

  http{{ if .tls }}s{{ end }}://{{ .name }}{{ .path | default "/" }}
{{- end }}

OR
{{- end }}

{{- if contains "NodePort" .Values.service.type }}

  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ $svc }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT{{ $url }}/

{{- else if contains "LoadBalancer" .Values.service.type }}

** Please ensure an external IP is associated to the {{ $svc }} service before proceeding **
** Watch the status using: kubectl get svc --namespace {{ .Release.Namespace }} -w {{ $svc }} **

  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ $svc }} -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
  echo http://$SERVICE_IP:{{ .Values.service.externalPort }}{{ $url }}/

OR

  export SERVICE_HOST=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ $svc }} -o jsonpath='{.status.loadBalancer.ingress[0].hostname}')
  echo http://$SERVICE_HOST:{{ .Values.service.externalPort }}{{ $url }}/

{{- else if contains "ClusterIP"  .Values.service.type }}

  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ template "stirlingpdf.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  echo http://127.0.0.1:8080{{ $url }}/
  kubectl port-forward $POD_NAME 8080:8080 --namespace {{ .Release.Namespace }}

{{- end }}
